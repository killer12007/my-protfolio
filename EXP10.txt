#include <stdio.h>
#include <stdlib.h>
#include <conio.h>

#define SIZE 10

// Node structure for chaining
typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Hash table for chaining
Node* chain[SIZE];

// Initialize hash table
void init() {
    int i;
    for (i = 0; i < SIZE; i++) {
        chain[i] = NULL;
    }
}

// Hash function
int hashFunction(int key) {
    return key % SIZE;
}

// Insertion using chaining
void insertChaining(int key) {
    int hash = hashFunction(key);
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = key;
    newNode->next = chain[hash];
    chain[hash] = newNode;
}

// Display chaining hash table
void displayChaining() {
    int i;
    for (i = 0; i < SIZE; i++) {
        Node* temp = chain[i];
        printf("[%d] -> ", i);
        while (temp) {
            printf("%d -> ", temp->data);
            temp = temp->next;
        }
        printf("NULL\n");
    }
}

// ---------------- OPEN ADDRESSING ----------------

int openAddressing[SIZE];

// Initialize open addressing hash table
void initOpenAddressing() {
    int i;
    for (i = 0; i < SIZE; i++) {
        openAddressing[i] = -1;
    }
}

// Insert using linear probing
void insertOpenAddressing(int key) {
    int hash = hashFunction(key);
    int i = 0;
    while (openAddressing[(hash + i) % SIZE] != -1) {
        i++;
    }
    openAddressing[(hash + i) % SIZE] = key;
}

// Display open addressing hash table
void displayOpenAddressing() {
    int i;
    for (i = 0; i < SIZE; i++) {
        printf("[%d] -> %d\n", i, openAddressing[i]);
    }
}

// Main function
int main() {
    clrscr();
    printf("Hash Table with Chaining:\n");
    init();
    insertChaining(5);
    insertChaining(15);
    insertChaining(25);
    insertChaining(35);
    displayChaining();

    printf("\nHash Table with Open Addressing (Linear Probing):\n");
    initOpenAddressing();
    insertOpenAddressing(5);
    insertOpenAddressing(15);
    insertOpenAddressing(25);
    insertOpenAddressing(35);
    displayOpenAddressing();

    getch();
    return 0;
}
