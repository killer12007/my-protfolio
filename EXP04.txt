#include <stdio.h>
#include <stdlib.h>
#include <conio.h>

#define MAX 5

// Queue using Array
typedef struct {
    int arr[MAX];
    int front, rear;
} ArrayQueue;

void initArrayQueue(ArrayQueue *q) {
    q->front = 0;
    q->rear = -1;
}
int isFullArray(ArrayQueue *q) {
    return q->rear == MAX - 1;
}
int isEmptyArray(ArrayQueue *q) {
    return q->front > q->rear;
}
void enqueueArray(ArrayQueue *q, int val) {
    if (isFullArray(q)) {
        printf("Queue Overflow!\n");
        return;
    }
    q->arr[++q->rear] = val;
}
int dequeueArray(ArrayQueue *q) {
    if (isEmptyArray(q)) {
        printf("Queue Underflow!\n");
        return -1;
    }
    return q->arr[q->front++];
}

// Circular Queue using Array
typedef struct {
    int arr[MAX];
    int front, rear;
} CircularQueue;

void initCircularQueue(CircularQueue *q) {
    q->front = q->rear = -1;
}
int isFullCircular(CircularQueue *q) {
    return (q->front == (q->rear + 1) % MAX);
}
int isEmptyCircular(CircularQueue *q) {
    return q->front == -1;
}
void enqueueCircular(CircularQueue *q, int val) {
    if (isFullCircular(q)) {
        printf("Circular Queue Overflow!\n");
        return;
    }
    if (q->front == -1) q->front = 0;
    q->rear = (q->rear + 1) % MAX;
    q->arr[q->rear] = val;
}
int dequeueCircular(CircularQueue *q) {
    int val;
    if (isEmptyCircular(q)) {
        printf("Circular Queue Underflow!\n");
        return -1;
    }
    val = q->arr[q->front];
    if (q->front == q->rear) {
        q->front = q->rear = -1;
    } else {
        q->front = (q->front + 1) % MAX;
    }
    return val;
}

// Queue using Linked List
typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node *front, *rear;
} LinkedQueue;

void initLinkedQueue(LinkedQueue *q) {
    q->front = q->rear = NULL;
}

void enqueueList(LinkedQueue *q, int val) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = val;
    newNode->next = NULL;
    if (q->rear == NULL) {
        q->front = q->rear = newNode;
    } else {
        q->rear->next = newNode;
        q->rear = newNode;
    }
}
int dequeueList(LinkedQueue *q) {
    Node* temp;
    int val;
    if (q->front == NULL) {
        printf("Queue Underflow!\n");
        return -1;
    }
    temp = q->front;
    val = temp->data;
    q->front = q->front->next;
    if (q->front == NULL) q->rear = NULL;
    free(temp);
    return val;
}

// Main function to demonstrate queues
int main() {
    ArrayQueue aq;
    CircularQueue cq;

    initCircularQueue(&cq);
    initLinkedQueue(&lq);

    printf("Demonstrating Array Queue:\n");
    enqueueArray(&aq, 10);
    enqueueArray(&aq, 20);
    printf("Dequeued: %d\n", dequeueArray(&aq));

    printf("\nDemonstrating Circular Queue:\n");
    enqueueCircular(&cq, 30);
    enqueueCircular(&cq, 40);
    printf("Dequeued: %d\n", dequeueCircular(&cq));

    printf("\nDemonstrating Linked List Queue:\n");
    enqueueList(&lq, 50);
    enqueueList(&lq, 60);
    printf("Dequeued: %d\n", dequeueList(&lq));

    getch();
    return 0;
}
