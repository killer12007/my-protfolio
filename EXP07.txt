#include <stdio.h>
#include <stdlib.h>
#include <conio.h>

// Singly Linked List Node
typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Doubly Linked List Node
typedef struct DNode {
    int data;
    struct DNode* prev;
    struct DNode* next;
} DNode;

// Circular Linked List Node
typedef struct CNode {
    int data;
    struct CNode* next;
} CNode;

// Functions for Singly Linked List
void insertSLL(Node** head, int data) {
    Node* newNode = (Node*)malloc(sizeof(Node));
    newNode->data = data;
    newNode->next = *head;
    *head = newNode;
}
void deleteSLL(Node** head) {
    Node* temp;
    if (*head == NULL) return;
    temp = *head;
    *head = (*head)->next;
    free(temp);
}
void displaySLL(Node* head) {
    while (head != NULL) {
        printf("%d -> ", head->data);
        head = head->next;
    }
    printf("NULL\n");
}

// Functions for Doubly Linked List
void insertDLL(DNode** head, int data) {
    DNode* newNode = (DNode*)malloc(sizeof(DNode));
    newNode->data = data;
    newNode->prev = NULL;
    newNode->next = *head;
    if (*head != NULL)
        (*head)->prev = newNode;
    *head = newNode;
}
void deleteDLL(DNode** head) {
    DNode* temp;
    if (*head == NULL) return;
    temp = *head;
    *head = (*head)->next;
    if (*head != NULL)
        (*head)->prev = NULL;
    free(temp);
}
void displayDLL(DNode* head) {
    while (head != NULL) {
        printf("%d <-> ", head->data);
        head = head->next;
    }
    printf("NULL\n");
}

// Functions for Circular Linked List
void insertCLL(CNode** head, int data) {
    CNode* newNode = (CNode*)malloc(sizeof(CNode));
    CNode* temp;
    newNode->data = data;
    if (*head == NULL) {
        newNode->next = newNode;
        *head = newNode;
        return;
    }
    temp = *head;
    while (temp->next != *head) {
        temp = temp->next;
    }
    temp->next = newNode;
    newNode->next = *head;
}
void deleteCLL(CNode** head) {
    CNode* temp;
    CNode* last;
    if (*head == NULL) return;
    if ((*head)->next == *head) {
        free(*head);
        *head = NULL;
        return;
    }
    temp = *head;
    last = *head;
    while (last->next != *head) {
        last = last->next;
    }
    *head = (*head)->next;
    last->next = *head;
    free(temp);
}
void displayCLL(CNode* head) {
    CNode* temp;
    if (head == NULL) return;
    temp = head;
    do {
        printf("%d -> ", temp->data);
        temp = temp->next;
    } while (temp != head);
    printf("(back to head)\n");
}

int main() {
    Node* sll = NULL;
    DNode* dll = NULL;
    CNode* cll = NULL;
    clrscr();

    printf("Singly Linked List Demo:\n");
    insertSLL(&sll, 10);
    insertSLL(&sll, 20);
    displaySLL(sll);
    deleteSLL(&sll);
    displaySLL(sll);

    printf("\nDoubly Linked List Demo:\n");
    insertDLL(&dll, 30);
    insertDLL(&dll, 40);
    displayDLL(dll);
    deleteDLL(&dll);
    displayDLL(dll);

    printf("\nCircular Linked List Demo:\n");
    insertCLL(&cll, 50);
    insertCLL(&cll, 60);
    displayCLL(cll);
    deleteCLL(&cll);
    displayCLL(cll);

    getch();
    return 0;
}
