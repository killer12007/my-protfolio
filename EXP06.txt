#include <stdio.h>
#include <time.h>
#include <conio.h>

#define MAX 100

void bubbleSort(int arr[], int n) {
    int i, j;
    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

void selectionSort(int arr[], int n) {
    int i, j;
    for (i = 0; i < n - 1; i++) {
        int minIndex = i;
        int temp;
        for (j = i + 1; j < n; j++) {
            if (arr[j] < arr[minIndex]) {
                minIndex = j;
            }
        }
        temp = arr[minIndex];
        arr[minIndex] = arr[i];
        arr[i] = temp;
    }
}

void insertionSort(int arr[], int n) {
    int i;
    for (i = 1; i < n; i++) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            j--;
        }
        arr[j + 1] = key;
    }
}

void printArray(int arr[], int n) {
    int i;
    for (i = 0; i < n; i++)
        printf("%d ", arr[i]);
    printf("\n");
}

int main() {
    int n, i;
    int arr[MAX], arrCopy[MAX];
    clock_t start, end;
    double timeTaken;

    clrscr();
    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
        arrCopy[i] = arr[i];
    }

    // Bubble Sort
    for (i = 0; i < n; i++) arrCopy[i] = arr[i];
    start = clock();
    bubbleSort(arrCopy, n);
    end = clock();
    timeTaken = ((double)(end - start)) / CLOCKS_PER_SEC;
    printf("Bubble Sort Result: ");
    printArray(arrCopy, n);
    printf("Time taken: %f seconds\n", timeTaken);
    // Selection Sort
    for (i = 0; i < n; i++) arrCopy[i] = arr[i];
    start = clock();
    selectionSort(arrCopy, n);
    end = clock();
    timeTaken = ((double)(end - start)) / CLOCKS_PER_SEC;
    printf("Selection Sort Result: ");
    printArray(arrCopy, n);
    printf("Time taken: %f seconds\n", timeTaken);

    // Insertion Sort
    for (i = 0; i < n; i++) arrCopy[i] = arr[i];
    start = clock();
    insertionSort(arrCopy, n);
    end = clock();
    timeTaken = ((double)(end - start)) / CLOCKS_PER_SEC;
    printf("Insertion Sort Result: ");
    printArray(arrCopy, n);
    printf("Time taken: %f seconds\n", timeTaken);

    getch();
    return 0;
}
